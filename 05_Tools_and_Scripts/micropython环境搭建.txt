1. 更新系统包管理器
sudo apt update
sudo apt upgrade -y

2. 安装编译工具链（ARM GCC）
sudo apt install -y build-essential libreadline-dev libffi-dev git
sudo apt install -y pkg-config
sudo apt install -y gcc-arm-none-eabi

3. 安装Python 3及pip
sudo apt install -y python3 python3-pip



#可选
4. 编译MicroPython交叉编译器（mpy-cross）
cd micropython
cd mpy-cross
make



5编译固件
cd ..
cd ports/stm32
make BOARD=STM32F413RGT6
